```gdscript
func _setup_resources_system() -> void:
    var api := Node.new()
    api.set_script(_ResourcesApiScript)
    api.name = "ResourcesApi"
    add_child(api)
    _resources_api = api

func _setup_resources_api_deferred() -> void:
```

```gdscript
## 检查并扣减建造资源。返回 {ok:true} 或 {ok:false, reason}
func _check_and_consume_cost(building_type: String, region_id: String) -> Dictionary:
    if _resources_api == null:
        return {"ok": true}  # 资源系统未接入，跳过检查
    # P0 硬编码费用映射（D1 阶段改为从 buildings.tres 读取）
    var costs: Dictionary = {
        "bld_workshop": {"res_wood": 120.0, "res_stone": 60.0, "res_metal_ore": 20.0},

    if not costs.has(building_type):
        return {"ok": true}  # 无费用定义，跳过
    var cost: Dictionary = costs[building_type]
    # 先检查
    for res_id in cost.keys():
        var stock: float = _resources_api.get_stock(res_id, region_id)
        if stock < cost[res_id]:
            return {"ok": false, "reason": "缺少 %s (需要 %d, 现有 %d)" % [res_id, cost[res_id], stock]}
    # 再扣减
    for res_id in cost.keys():
        var result: Dictionary = _resources_api.consume(res_id, cost[res_id], region_id, "建造:%s" % building_type)
        if not result.get("ok", false):
            return {"ok": false, "reason": "扣减失败: %s" % result.get("error", "")}
    return {"ok": true}
```

5. 在 `game_root.gd._setup_resources_api_deferred()` 之后追加注入逻辑（或在 `_setup_construction_api_deferred` 中追加）：

```gdscript
func _setup_construction_api_deferred() -> void:
    if _construction_api == null or _construction_manager == null:
        return
    if not _construction_api.has_method("setup"):
        return
    _construction_api.setup(_construction_manager)
    # P0-9 注入 ResourcesApi 到 ConstructionManager
    if _resources_api != null and _construction_manager.has_method("set_resources_api"):
        _construction_manager.set_resources_api(_resources_api)
```

> **注意**：`_setup_construction_api_deferred` 和 `_setup_resources_api_deferred` 都用 `call_deferred`，执行顺序按 `_ready` 中调用顺序。`_setup_resources_system()` 在 `_setup_construction_system()` 之后调用，但 deferred 调用是按注册顺序入队，所以 `_setup_construction_api_deferred` 会先执行。需要确保 ConstructionManager 注入 ResourcesApi 时后者已就绪。
>
> **解决方案**：把注入逻辑放到 `_setup_resources_api_deferred()` 末尾（此时 ResourcesApi 已就绪，ConstructionManager 也已在更早的 deferred 中就绪）：

```gdscript
func _setup_resources_api_deferred() -> void:
    if _resources_api == null:
        return
    if not _resources_api.has_method("setup"):
        return
    var mgr = _ResourcesManagerScript.new()
    _resources_api.setup(mgr)
    # 注入到 ConstructionManager（P0-9）
    if _construction_manager != null and _construction_manager.has_method("set_resources_api"):
        _construction_manager.set_resources_api(_resources_api)
```

**EventBus 信号**：`event_bus.gd:24` 已声明 `resource_changed(resource_name, amount, delta)` 信号。`resources/api.gd` 的 `consume()`/`produce()` 会发射本地 `resource_changed` 信号，但**不会自动转发到 EventBus**。如需全局监听，需在 `_setup_resources_api_deferred` 中连接：

```gdscript
if EventBus != null and _resources_api.has_signal("resource_changed"):
    _resources_api.resource_changed.connect(func(rid, amt, delta, rid2):
        EventBus.resource_changed.emit(rid, int(amt), int(delta))
    )
```

> 注意 api.gd 的信号签名是 `(resource_id, amount, delta, region_id)`，EventBus 是 `(resource_name, amount, delta)`，签名不完全匹配，C1 阶段先不转发，C2 如需全局监听再处理。
